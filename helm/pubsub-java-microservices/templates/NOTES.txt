1. Verify all services are running:
   kubectl get pods -n {{ .Release.Namespace }}

2. Check service status:
   kubectl get svc -n {{ .Release.Namespace }}

3. View logs:
   # Producer
   kubectl logs -f deployment/producer -n {{ .Release.Namespace }}
   
   # Order Processing
   kubectl logs -f deployment/order-processing -n {{ .Release.Namespace }}
   
   # Notification
   kubectl logs -f deployment/notification -n {{ .Release.Namespace }}

4. Access services (port-forward):
   # Producer
   kubectl port-forward --namespace {{ .Release.Namespace }} svc/producer 8080:8080
   # Health check: curl http://localhost:8080/actuator/health
   
   # Order Processing
   kubectl port-forward --namespace {{ .Release.Namespace }} svc/order-processing 8081:8080
   # Health check: curl http://localhost:8081/actuator/health
   
   # Notification
   kubectl port-forward --namespace {{ .Release.Namespace }} svc/notification 8082:8080
   # Health check: curl http://localhost:8082/actuator/health
